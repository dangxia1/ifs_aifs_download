## 四、测速（按需）

```bash
cd /home/zongshen/ifs_aifs_download/dl_lastmonth_cal_ivt
conda activate ifs_aifs

python -c "
from ecmwf.opendata import Client
import time, os
for src in ['aws', 'azure']:
    t0 = time.time()
    c = Client(source=src, model='ifs')
    c.retrieve(date='2026-06-15', time=0, type='fc', step=0,
               param=['tp'], target=f'/tmp/_{src}.grib2')
    sz = os.path.getsize(f'/tmp/_{src}.grib2')
    print(f'{src:>6s}: {time.time()-t0:5.1f}s  ({sz} bytes)')
    os.remove(f'/tmp/_{src}.grib2')
"
```

## 五、运行

### 5.1 dl_realtime（实时预报 + IVT + 可视化）

```bash
cd /home/zongshen/ifs_aifs_download/dl_realtime
conda activate ifs_aifs

# 后台跑

#   9 个 .pdf    → ec_realtime/figures/{ifs,aifs}/
```

```bash

# dl_realtime
kill $(ps aux | grep dl_realtime | grep -v grep | awk '{print $2}')
```
